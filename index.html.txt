<body>
  <div class="container" id="container">
    <!-- SVG Logo图片 -->
    <img src="logo.svg" alt="yi-on" class="logo">
    <div class="instruction">任意圆形，一按出结果</div>
  </div>

  <script>
    const container=document.getElementById('container');
    const circleCount=6;
    const colors=['color-1','color-2','color-3','color-4','color-5','color-6'];
    let circles=[];

    // 生成唯一用户ID的函数
    function genUID(){return 'U'+Math.random().toString(36).slice(2,10);}
    
    // 数组洗牌函数：随机重新排列数组元素
    function shuffleArray(a){for(let i=a.length-1;i>0;i--){const j=Math.floor(Math.random()*(i+1));[a[i],a[j]]=[a[j],a[i]];}return a;}

    // 创建圆形元素的函数
    function createCircles(){
      // 移除所有已存在的圆形并清空数组
      circles.forEach(c=>c.remove());circles=[];
      
      // 生成不同大小的圆形尺寸数组并随机排序
      // 尺寸基于容器尺寸计算，逐渐减小
      const sizes=shuffleArray(Array.from({length:circleCount},(_,i)=>Math.min(container.offsetWidth,container.offsetHeight)*(1-i*.1)*.25));
      
      // 存储已放置圆形的位置信息
      const placed=[];
      // 随机分配颜色
      const colorAs=shuffleArray([...colors]);
      
      // 循环创建每个圆形
      for(let i=0;i<circleCount;i++){
        // 创建圆形div元素
        const c=document.createElement('div');
        // 添加基础类和颜色类
        c.classList.add('circle',colorAs[i]);
        // 随机设置动画类型（1-3）
        c.setAttribute('data-animation',(Math.floor(Math.random()*3)+1));
        
        // 获取当前圆形的尺寸和随机动画持续时间
        const size=sizes[i],duration=7+Math.random()*2;
        
        // 设置CSS自定义属性：动画持续时间和动画类型
        c.style.setProperty('--breathe-duration',`${duration}s`);
        c.style.setProperty('--breathe-animation',`breathe${Math.floor(Math.random()*3)+1}`);
        
        // 查找合适的位置放置圆形（避免重叠）
        const pos=findPos(size,placed);
        
        // 设置圆形尺寸和位置
        c.style.width=c.style.height=`${size}px`;
        c.style.left=`${pos.x}px`;c.style.top=`${pos.y}px`;
        
        // 设置圆形ID
        c.dataset.id=i+1;
        
        // 添加点击事件监听器
        c.addEventListener('click',(e)=>{
          // 获取当前时间戳
          const ts=Date.now();
          // 获取或生成用户ID
          const uid=localStorage.getItem('zen_uid')||genUID();
          // 如果本地存储中没有用户ID，则保存
          if(!localStorage.getItem('zen_uid'))localStorage.setItem('zen_uid',uid);
          
          // 获取客户端IP地址（如果可用）
          const ip=window.__ZEN_TAP_CLIENT_IP||'0.0.0.0';
          // 获取点击坐标
          const x=Math.round(e.clientX),y=Math.round(e.clientY);
          
          // 生成种子字符串，包含时间、IP、用户ID、圆形ID和点击位置
          const seed=`${ts}|${ip}|${uid}|${c.dataset.id}|${x},${y}`;
          
          // 点击动画效果：放大并淡出
          c.style.transform='scale(2.5)';c.style.opacity='0';
          
          // 延迟500毫秒后跳转到内容页面，传递种子参数
          setTimeout(()=>location.href=`content.html?seed=${encodeURIComponent(seed)}`,500);
        });
        
        // 记录当前圆形的位置信息
        placed.push({x:pos.x,y:pos.y,radius:size/2});
        
        // 将圆形添加到容器并记录到数组
        container.appendChild(c);circles.push(c);
      }
    }
    
    // 查找合适位置放置圆形的函数
    function findPos(size,placed){
      const maxA=200; // 最大尝试次数
      let a=0; // 当前尝试次数
      const radius=size/2; // 圆形半径
      
      // 循环尝试找到不重叠的位置
      while(a<maxA){
        // 生成随机位置（确保圆形完全在容器内）
        const x=Math.random()*(container.offsetWidth-size),y=Math.random()*(container.offsetHeight-size);
        
        // 检查是否与所有已放置圆形都不重叠（保持10px间距）
        if(placed.every(p=>Math.sqrt((p.x-x)**2+(p.y-y)**2)>=(p.radius+radius+10)))return{x,y};
        a++;
      }
      
      // 如果尝试次数用完仍未找到合适位置，返回随机位置
      return{x:Math.random()*(container.offsetWidth-size),y:Math.random()*(container.offsetHeight-size)};
    }
    
    // 窗口大小改变时重新创建圆形（响应式布局）
    window.addEventListener('resize',createCircles);
    
    // 键盘事件：按R键重新创建圆形
    document.addEventListener('keydown',e=>e.key.toLowerCase()==='r'&&createCircles());
    
    // 页面加载完成后初始化创建圆形
    document.addEventListener('DOMContentLoaded',createCircles);
  </script>
