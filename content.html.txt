  <script>
    // 同频统计 
    const virtualCount = Math.floor(Math.random() * 12000) + 8000;
    document.getElementById('syncCount').textContent = virtualCount.toLocaleString();

    (async function(){
      const params = new URLSearchParams(location.search);
      const seed = params.get('seed');
      if(!seed) {
        document.getElementById('sections').innerHTML = '<p style="color:#8b0000; text-align:center;">缺少参数，请返回首页重新点击</p>';
        return;
      }
      
      try {
        const [ts, ip, uid, circleId, xy] = seed.split('|');
        const result = await ZenTap.generateHexagramResult({timestamp: Number(ts), ip: ip, uid: uid});
        const seg = result.main.segments;
        
        document.getElementById('hexId').textContent = result.summary;
        
        // 填充内容到各个卡片
        document.getElementById('statusContent').innerHTML = `<p>${seg.status}</p>`;
        document.getElementById('trendContent').innerHTML = `<p>${seg.trend}</p>`;
        document.getElementById('mindContent').innerHTML = `<p>${seg.mind}</p>`;
        document.getElementById('riskContent').innerHTML = `<p>${seg.risk}</p>`;
        
      } catch(e) {
        document.getElementById('sections').innerHTML = `<p style="color:#8b0000; text-align:center;">读取失败：${e.message}</p>`;
        console.error(e);
      }
    })();
  </script>
